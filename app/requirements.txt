# Optiseek dependencies — installed on top of the ROCm base image.
#
# The base image (rocm/pytorch:rocm10.0_ubuntu26.04_py3.14_pytorch_release_2.13.0)
# already ships: torch, PIL/Pillow, numpy. We add only what we need.
#
# IMPORTANT: The base image uses Python 3.14, which is very new. Older vLLM
# versions (0.8.x, 0.9.x, 0.10.x-0.19.x) require Python <3.13 or <3.14 and
# will FAIL to install. We use vllm>=0.20.0 which is the first line to
# support Python 3.14.
#
# If vLLM ROCm wheels are unavailable for Python 3.14 on your hardware,
# set OPTISEEK_ENGINE=transformers to skip vLLM and use the transformers
# fallback path (slower but reliable).

# Inference — vLLM with ROCm support (primary engine, optional)
# Use a range so pip can pick the latest Python-3.14-compatible build.
vllm>=0.20.0

# Transformers — used both by vLLM internally AND as our fallback engine.
# 4.49+ supports Qwen2.5-VL on Python 3.14.
transformers>=4.49.0

# Tokenizer / processor support (usually pulled in by transformers, but pinned)
tokenizers>=0.21.0
accelerate>=1.2.0

# Image handling — Pillow is in base but we pin for safety
Pillow>=10.4.0

# HF Hub — for downloading weights at runtime if not baked into /models
huggingface_hub>=0.27.0

# Logging / utilities — lightweight
# (nothing extra needed; stdlib logging is enough)
